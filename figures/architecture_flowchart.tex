\node[question] (grid) {Data on a regular grid or sequence?};
\node[answer, right=14mm of grid] (cnn) {CNN / RNN / Transformer\\ (\S~\ref{subsec:bp-when-not})};

\node[answer, right=14mm of edges] (four) {choose: set, complete, or learned\\ (\S~\ref{subsec:no-adjacency})};

\draw[arrow] (impose) -- (msg);
\draw[arrow] (msg) -- node[lab] {no} (sage);
\draw[arrow] (msg) -- node[lab] {yes} (mp);

\draw[arrow, black!50, dashed] (mp.south) to[out=-90, in=10] (start.east);
